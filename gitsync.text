# ─────────────────────────────────────────
# Git setup on a fresh PC
# ─────────────────────────────────────────

git config --global user.name "gtztemp"
git config --global user.email "gaurav.rockins@gmail.com"
git config --global core.fileMode false

git init
git branch -M main
git remote add origin https://github.com/gtztemp/Workspace.git
git remote -v

git fetch origin

# ─────────────────────────────────────────
# Choose ONE:
#
# If Workspace files were manually copied
# from the previous PC:
# ─────────────────────────────────────────

git reset --mixed origin/main

# This keeps the current local files and
# makes local main follow GitHub's history.
#
# ─────────────────────────────────────────
# OR
#
# If Workspace is empty / no local files:
# ─────────────────────────────────────────

git merge origin/main

# ─────────────────────────────────────────

git status

git add .
git commit -m "synced"
git push

# Then configure your shortcut

git config --global alias.sync '!git add . && git commit -m "synced" || true && git push'
git sync

// on another pc
    git pull
    git config --global alias.sync '!git add . && git commit -m "synced" || true && git push'
    git sync

// revert git add .
    git restore --cached .

// revert git commit 
    git reset --soft HEAD~1
    
// revert commit and add .
    git reset HEAD~1
    
// I want to go back to Stage, where I had started working today (soft reset , wont modify local directory changes) 
    git reset HEAD~1
    git push --force-with-lease
    
// I want to go back to Stage, where I had started working today (hard reset) -
    git reset --hard HEAD~1
    git push --force-with-lease    

// Monthly / occasional maintenance
    git fetch --prune && git gc

If you added very large files and want immediate removal -
    git reflog expire --expire=now --all 
    git gc --prune=now

// Check repository health
    git fsck
